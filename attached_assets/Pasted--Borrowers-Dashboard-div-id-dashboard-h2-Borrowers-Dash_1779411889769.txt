<!-- Borrowers Dashboard -->
<div id="dashboard">
  <h2>Borrowers Dashboard</h2>
  <table class="excel-style">
    <thead>
      <tr>
        <th>ID</th>
        <th>Name</th>
        <th>Total Loans</th>
        <th>Outstanding Balance</th>
        <th>Actions</th>
      </tr>
    </thead>
    <tbody id="dashboardBody"></tbody>
  </table>
</div>

<script>
  // ================================
  // 1) Load borrowers from localStorage
  // ================================
  let borrowers = [];

  function loadBorrowers() {
    const saved = localStorage.getItem("borrowers");
    borrowers = saved ? JSON.parse(saved) : [];
  }

  function saveBorrowers() {
    localStorage.setItem("borrowers", JSON.stringify(borrowers));
  }

  // ================================
  // 2) Add new borrower (only once)
  // ================================
  function addBorrower(name) {
    const id = Date.now();
    const borrower = { id, name, loans: [] };
    borrowers.push(borrower);
    saveBorrowers();
    renderDashboard();
  }

  // ================================
  // 3) Add new loan to existing borrower
  // ================================
  function addLoan(borrowerId, amount, date, interestRate) {
    const borrower = borrowers.find(b => b.id === borrowerId);
    if (!borrower) return;

    borrower.loans.push({
      loanId: Date.now(),
      amount,
      date,
      interestRate,
      balance: amount,
      payments: []
    });

    saveBorrowers();
    renderDashboard();
  }

  // ================================
  // 4) Render dashboard
  // ================================
  function renderDashboard() {
    const tbody = document.getElementById("dashboardBody");
    tbody.innerHTML = "";

    if (borrowers.length === 0) {
      tbody.innerHTML = `<tr><td colspan="5" style="text-align:center;">No borrowers yet.</td></tr>`;
      return;
    }

    borrowers.forEach((b, index) => {
      const totalLoans = b.loans.length;
      const outstanding = b.loans.reduce((sum, loan) => sum + loan.balance, 0);

      const row = document.createElement("tr");
      row.innerHTML = `
        <td>${index + 1}</td>
        <td>${b.name}</td>
        <td>${totalLoans}</td>
        <td>MOP ${outstanding}</td>
        <td>
          <button onclick="viewBorrower(${b.id})">View</button>
        </td>
      `;
      tbody.appendChild(row);
    });
  }

  // ================================
  // 5) Example: View borrower details
  // ================================
  function viewBorrower(borrowerId) {
    const borrower = borrowers.find(b => b.id === borrowerId);
    if (!borrower) return;

    alert(`Borrower: ${borrower.name}\nLoans: ${borrower.loans.length}`);
  }

  // ================================
  // 6) Initialize
  // ================================
  loadBorrowers();
  renderDashboard();
</script>
