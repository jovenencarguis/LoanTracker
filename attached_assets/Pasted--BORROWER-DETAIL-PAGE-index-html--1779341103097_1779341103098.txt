/* ============================================
   BORROWER DETAIL JS (script.js)
   ============================================ */

function renderBorrowerDetail(borrowerId) {
  const borrower = borrowers.find(b => b.id === borrowerId);
  if (!borrower) return;

  currentBorrowerId = borrowerId;

  document.getElementById("borrowerDetail").style.display = "block";
  document.getElementById("borrowerList").style.display = "none";
  document.getElementById("newBorrowerForm").style.display = "none";
  document.getElementById("editBorrowerForm").style.display = "none";
  document.getElementById("addPaymentForm").style.display = "none";

  const info = `
    <div>
      <strong>Name:</strong> ${borrower.name}<br>
      <strong>Starting Balance:</strong> MOP ${borrower.startingBalance}<br>
      <strong>Current Balance:</strong> MOP ${borrower.currentBalance}<br>
      <strong>Interest Rate:</strong> ${borrower.interestRate}%<br>
      <strong>Date Borrowed:</strong> ${borrower.dateBorrowed}<br>
      <strong>Notes:</strong> ${borrower.notes || "None"}
    </div>
  `;

  document.getElementById("borrowerInfo").innerHTML = info;

  renderPaymentHistory(borrower);
}



/* ============================================
   PAYMENT HISTORY TABLE JS (script.js)
   ============================================ */

function renderPaymentHistory(borrower) {
  const tbody = document.getElementById("paymentHistoryBody");
  tbody.innerHTML = "";

  if (borrower.payments.length === 0) {
    tbody.innerHTML = `
      <tr><td colspan="8" style="text-align:center;">No payments recorded yet.</td></tr>
    `;
    return;
  }

  borrower.payments.forEach(payment => {
    const typeLabel = payment.repayment > 0
      ? "Principal + Interest"
      : "Interest‑Only";

    const row = document.createElement("tr");

    row.innerHTML = `
      <td>${payment.date}</td>
      <td>${typeLabel}</td>
      <td>MOP ${payment.previousBalance}</td>
      <td>MOP ${payment.interest}</td>
      <td>MOP ${payment.repayment}</td>
      <td>MOP ${payment.totalCollected}</td>
      <td>MOP ${payment.newBalance}</td>
      <td>
        <button class="delete-btn" onclick="deletePayment(${borrower.id}, ${payment.id})">
          Delete
        </button>
      </td>
    `;

    tbody.appendChild(row);
  });
}
